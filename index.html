${cat.name}
가치 × 별 = ${data.val * data.star}점
`; }).join(''); return `
${players.length > 1 ? `` : ''}
${catHtml}
석재 보너스
총점 ${calculateTotal(player)}점
`; }).join(''); } // 4. 전역 함수 수정 (전체 리렌더링 제거) window.updateName = (id, name) => { players.find(p => p.id === id).name = name; renderRanking(); }; window.updateValue = (id, catId, field, val) => { const player = players.find(p => p.id === id); player.values[catId][field] = Number(val) || 0; // 전체를 다시 그리지 않고 특정 요소만 업데이트 const catResult = player.values[catId].val * player.values[catId].star; document.getElementById(`res-${id}-${catId}`).innerText = `= ${catResult}점`; document.getElementById(`total-${id}`).querySelector('span:last-child').innerText = `${calculateTotal(player)}점`; renderRanking(); // 랭킹은 다시 그릴 수 있음 (입력창과 무관함) }; window.updateStone = (id, val) => { const player = players.find(p => p.id === id); player.stone = Number(val) || 0; document.getElementById(`total-${id}`).querySelector('span:last-child').innerText = `${calculateTotal(player)}점`; renderRanking(); }; window.deletePlayer = (id) => { players = players.filter(p => p.id !== id); renderPlayers(); // 삭제 시에는 전체 갱신 필요 renderRanking(); }; // [5. 버튼 이벤트 등 나머지 코드는 기존과 동일]

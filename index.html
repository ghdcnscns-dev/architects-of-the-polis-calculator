// [이전 코드 생략: 변수 정의 부분은 그대로 유지] // 3. 플레이어 카드 렌더링 (입력 시 전체 리렌더링 방지) function renderPlayers() { playersContainer.innerHTML = players.map(player => { let catHtml = categories.map(cat => { const data = player.values[cat.id]; // ID를 고유하게 생성하여 DOM 업데이트가 아닌 값만 변경되도록 유도 return `
${cat.name}
⭐별 × ⬡타일 = ${data.star * data.val}점
`; }).join(''); return `
${players.length > 1 ? `` : ''}
${catHtml}
석재 보너스
총점 ${calculateTotal(player)}점
`; }).join(''); } // 4. 개별 값만 업데이트 (전체 리렌더링 제거) window.updateValue = (id, catId, field, val, inputEl) => { const player = players.find(p => p.id === id); player.values[catId][field] = Number(val) || 0; // 1. 해당 구역의 점수만 즉시 계산 및 업데이트 const catResult = player.values[catId].star * player.values[catId].val; document.getElementById(`res-${id}-${catId}`).innerText = `= ${catResult}점`; // 2. 전체 총점 업데이트 document.getElementById(`total-${id}`).querySelector('span:last-child').innerText = `${calculateTotal(player)}점`; // 3. 랭킹만 갱신 (전체 카드는 다시 그리지 않음) renderRanking(); }; window.updateStone = (id, val) => { const player = players.find(p => p.id === id); player.stone = Number(val) || 0; document.getElementById(`total-${id}`).querySelector('span:last-child').innerText = `${calculateTotal(player)}점`; renderRanking(); }; // [삭제/추가 함수 등은 기존 그대로 유지]
